interface PropertyCardProps {
  listingData: LocalizedPropertyPayload;
  activeLanguage: Language;
  isSaved: boolean;
  onLanguageChange: (lang: Language) => void;
  onToggleSave: (id: string) => void;
}

export const PropertyCard: React.FC<PropertyCardProps> = ({
  listingData,
  activeLanguage,
  isSaved,
  onLanguageChange,
  onToggleSave,
}) => {
  const handleOpenSource = async (url: string) => {
    try {
      const supported = await Linking.canOpenURL(url);
      if (supported) {
        await Linking.openURL(url);
      } else {
        Alert.alert('Error', `Cannot open URL: ${url}`);
      }
    } catch {
      Alert.alert('Error', 'Failed to open source link.');
    }
  };

  return (
    <View style={styles.card}>
      <View style={styles.cardHeader}>
        <View style={styles.buttonGroup}>
          {LANGUAGES.map((lang) => {
            const isActive = activeLanguage === lang;
            return (
              <TouchableOpacity
                key={lang}
                accessibilityRole="button"
                accessibilityState={{ selected: isActive }}
                accessibilityLabel={`Switch to ${lang.toUpperCase()}`}
                style={[styles.langButton, isActive && styles.activeLang]}
                onPress={() => onLanguageChange(lang)}
              >
                <Text style={[styles.langText, isActive && styles.activeLangText]}>
                  {lang.toUpperCase()}
                </Text>
              </TouchableOpacity>
            );
          })}
        </View>

        <TouchableOpacity
          accessibilityRole="button"
          accessibilityLabel={isSaved ? 'Remove from saved' : 'Save property'}
          style={[styles.bookmarkButton, isSaved && styles.bookmarkButtonActive]}
          onPress={() => onToggleSave(listingData.id)}
        >
          <Text style={styles.bookmarkText}>{isSaved ? '♥' : '♡'}</Text>
        </TouchableOpacity>
      </View>

      <Text style={styles.title} numberOfLines={2}>
        {listingData.title}
      </Text>
      <Text style={styles.price}>{listingData.price_formatted}</Text>
      <Text style={styles.details}>{listingData.details}</Text>
      <Text style={styles.address}>{listingData.location.address}</Text>

      <View style={styles.badgeContainer}>
        {Boolean(listingData.commute_info) && (
          <View style={styles.badge}>
            <Text style={styles.badgeText}>{listingData.commute_info}</Text>
          </View>
        )}
        {Boolean(listingData.tax_info) && (
          <View style={styles.badge}>
            <Text style={styles.badgeText}>{listingData.tax_info}</Text>
          </View>
        )}
      </View>

      {listingData.sources && listingData.sources.length > 0 && (
        <View style={styles.sourcesContainer}>
          <Text style={styles.sourcesLabel}>{listingData.sources_label}:</Text>
          <View style={styles.sourcePills}>
            {listingData.sources.map((src, idx) => (
              <TouchableOpacity
                key={`${src.portal}-${idx}`}
                accessibilityRole="link"
                accessibilityLabel={`Open ${src.portal} source`}
                style={styles.sourceLink}
                onPress={() => handleOpenSource(src.url)}
              >
                <Text style={styles.sourceLinkText}>{src.portal} ↗</Text>
              </TouchableOpacity>
            ))}
          </View>
        </View>
      )}
    </View>
  );
};

// 5. TAB SCREENS
// ==========================================

function SearchScreen() {
  const [globalLanguage, setGlobalLanguage] = useState<Language>('de');
  const [searchQuery, setSearchQuery] = useState<string>('');
  const [selectedCanton, setSelectedCanton] = useState<string>('ALL');
  const [savedIds, setSavedIds] = useState<Set<string>>(new Set());

  const i18n = UI_I18N[globalLanguage];

  const handleToggleSave = (id: string) => {
    setSavedIds((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  };

  const filteredListings = useMemo(() => {
    return Object.values(PROPERTY_DATABASE)
      .map((item) => item[globalLanguage])
      .filter((property) => {
        const matchesCanton = selectedCanton === 'ALL' || property.location.canton === selectedCanton;
        const matchesSearch =
          property.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
          property.location.city.toLowerCase().includes(searchQuery.toLowerCase()) ||
          property.location.address.toLowerCase().includes(searchQuery.toLowerCase());
        return matchesCanton && matchesSearch;
      });
  }, [globalLanguage, selectedCanton, searchQuery]);

  return (
    <SafeAreaView style={styles.screen}>
      <View style={styles.searchContainer}>
        <TextInput
          style={styles.searchInput}
          placeholder={i18n.searchPlaceholder}
          placeholderTextColor="#8E8E93"
          value={searchQuery}
          onChangeText={setSearchQuery}
        />
      </View>

      <View style={styles.filterRow}>
        {['ALL', 'ZH', 'GE', 'TI'].map((canton) => (
          <TouchableOpacity
            key={canton}
            style={[styles.filterChip, selectedCanton === canton && styles.filterChipActive]}
            onPress={() => setSelectedCanton(canton)}
          >
            <Text style={[styles.filterChipText, selectedCanton === canton && styles.filterChipTextActive]}>
              {canton === 'ALL' ? i18n.allCantons : canton}
            </Text>
          </TouchableOpacity>
        ))}
      </View>

      <ScrollView contentContainerStyle={styles.feedContainer}>
        {filteredListings.length > 0 ? (
          filteredListings.map((property) => (
            <PropertyCard
              key={property.id}
              listingData={property}
              activeLanguage={globalLanguage}
              isSaved={savedIds.has(property.id)}
              onLanguageChange={setGlobalLanguage}
              onToggleSave={handleToggleSave}
            />
          ))
        ) : (
          <View style={styles.emptyState}>
            <Text style={styles.emptyStateText}>{i18n.noResults}</Text>
          </View>
        )}
      </ScrollView>
    </SafeAreaView>
  );
}

// 7. MAIN APP ENTRY WITH SPLASH SCREEN
// ==========================================

export default function App() {
  const [appIsReady, setAppIsReady] = useState(false);

  useEffect(() => {
    async function prepare() {
      try {
        await new Promise((resolve) => setTimeout(resolve, 1500));
      } catch (e) {
        console.warn(e);
      } finally {
        setAppIsReady(true);
      }
    }
    prepare();
  }, []);

  const onLayoutRootView = useCallback(async () => {
    if (appIsReady) {
      await SplashScreen.hideAsync();
    }
  }, [appIsReady]);

  if (!appIsReady) {
    return (
      <View style={styles.splashContainer}>
        <StatusBar barStyle="light-content" backgroundColor="#DA291C" />
        <HelvetiCasaLogo size={120} />
        <Text style={styles.splashTitle}>
          Helveti<Text style={styles.splashAccent}>Casa</Text>
        </Text>
        <Text style={styles.splashTagline}>SWISS REAL ESTATE</Text>
      </View>
    );
  }

  return (
    <NavigationContainer onReady={onLayoutRootView}>
      <StatusBar barStyle="dark-content" />
      <TabNavigator />
    </NavigationContainer>
  );
}
